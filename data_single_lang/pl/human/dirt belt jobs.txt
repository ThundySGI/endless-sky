mission "Drought Relief"
	name `Pomoc w związku z suszą do <planet>`
	job
	repeat
	description `Świat rolnictwa <destination> doświadcza obecnie suszy. Dostarcz <cargo> na planetę do <date> za <payment>.`
	cargo "drought relief supplies" 25 2 .05
	deadline
	to offer
		random < 10
	source
		government "Republic"
		not attributes "farming"
	destination
		distance 3 10
		attributes "dirt belt"
		attributes "farming"
	on visit
		dialog phrase "generic cargo on visit"
	on complete
		payment 10000 160
		dialog "Zrzucasz ładunek <commodity> i odbierasz zapłatę w wysokości <payment>. Grupa <planet> obywateli dziękuje ci i żegna cię."


mission "To Remembrance Day celebration [0]"
	name `Obchody Dnia Pamięci`
	job
	repeat
	description `Przynieś <fare> do <destination> do <date> na obchody Dnia Pamięci 19 lutego. Zapłata wynosi <payment>.`
	passengers 4 4 .5
	deadline
	to offer
		or
			and
				random < 20
				month == 12
				day >= 21
			and
				random < 30
				month == 1
			and
				random < 10
				month == 2
				day < 19
	to fail
		month * 100 + day >= 219
	source
		attributes "dirt belt"
	destination
		distance 3 15
		attributes "dirt belt"
	on visit
		dialog phrase "generic passenger on visit"
	on complete
		payment 1000 180
		dialog "Życzysz swojemu <passengers> wszystkiego najlepszego na <planet>. Dzieci uciekają z twojego statku pomalowane na zielono, a rodzice płacą ci <payment>."
