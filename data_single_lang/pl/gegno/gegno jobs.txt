mission "Gegno Asteroid Mining - Aluminum"
	job
	repeat
	name "Zbierz 10 ton aluminium"
	description "Wydobądź i zbierz 10 ton aluminium, a następnie przywieź minerały do Gegno w punkcie <origin>."
	source "Tschyss"
	to offer
		has "Gegno Asteroid Mining Prologue: done"
		random < 10
	to accept
		"cargo space" > 10
	on visit
		dialog "Wróciłeś do <origin>, ale nie masz 10 ton aluminium w ładowni. Upewnij się, że wydobyłeś 10 ton aluminium i zapakowałeś je do ładowni przed lądowaniem."
	on complete
		"gegno: mining jobs" ++
		payment 36000
		outfit "Aluminum" -10
		dialog phrase "gegno mining payment dialog"
