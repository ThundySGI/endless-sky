mission "Unfettered Jump Drive 4"
	job
	repeat
	name "Sprzedaj napęd rozruchowy"
	description "Wymień napęd rozruchowy na swoim okręcie flagowym lub w ładunku na pieniądze lub broń, jeśli dostępny jest dostawca."
	source
		attributes "unfettered"
	to offer
		has "Unfettered Jump Drive 3: offered"
		not "event: wanderers: unfettered invasion starts"
	to fail
		has "Unfettered Jump Drive 4: active"
	to accept
		or
			has "outfit (flagship installed): Jump Drive"
			has "outfit (cargo): Jump Drive"
	on accept
		conversation
			branch "want outfits"
				not "unfettered: money for drives"
				has "Unfettered Jump Drive Trading: offered"
				has "flagship planet: Darkcloak"
			label money
			action
				payment 1000000
			`Jak zwykle, Unfettered są bardziej niż skłonni zapłacić ci milion kredytów za twój napęd skokowy, ale nie uzyskasz żadnych dodatkowych informacji rozmawiając z nimi.`
				goto trade

			label "want outfits"
			`Zwyczajowa umowa Unfettered, polegająca na wymianie napędu skokowego na broń, nadal obowiązuje.`
			choice
				`	(Weź dwa niszczarki Tripulse.)`
				`	(Weź zamiast tego pieniądze.)`
					goto money
			action
				outfit "Tripulse Shredder" 2
			`	Dwa Shreddery są załadowane na twój statek w zamian za napęd skokowy.`

			label trade
			branch spareJD
				"outfit (cargo): Jump Drive" + "outfit (flagship installed): Hyperdrive" + "outfit (flagship installed): Scram Drive" + "outfit (flagship installed): Jump Drive" > 1
			action
				outfit "Hyperdrive" 1
			label spareJD
			action
				outfit "Jump Drive" -1
				"reputation: Hai (Unfettered)" >?= 40
